//Brazil

//print('Band names:', bands_Brazil_col10);

var bandName = "classification_" + anio;

// Get unique values
var uniqueValues = ee.Dictionary(
  Brazil_col10_area.select(bandName)
    .reduceRegion({
      reducer: ee.Reducer.frequencyHistogram(),
      geometry: Brazil_col10_area.geometry(),
      scale: 30,
      maxPixels: 1e13,
      bestEffort: true
    })
    .get(bandName)
).keys();

print('Unique values for Brazil ' + bandName + ':', uniqueValues);


var origen_BR = [1,3,4,6,49,10,11,12,29,50,15,14,18,39,20,40,62,41,36,46,47,35,48,9,21,22,23,24,30,75,25,26,33,31,27]

var images_remapped = Brazil_col10_area.bandNames().map(function(band){
  return Brazil_col10_area.select([band]).remap(origen_BR, destino_BR).rename([band])
})

var i_outputs_BR =  ee.ImageCollection.fromImages(images_remapped).toBands()
      
i_outputs_BR = i_outputs_BR.select(i_outputs_BR.bandNames(),bands_BR)
            .set({
              version: "reclass",
              years: "1985-2024"
            }).select(bands_BR.removeAll(["classification_1984"]))

print(i_outputs_BR, "BR")
var BR_area_remap = i_outputs_BR.uint8()
